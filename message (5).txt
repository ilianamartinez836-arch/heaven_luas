-- FPS Boost Script - Ported to new API
local original_cvars = {}

-- Helper functions for setting CVars with original value tracking
local function SetI(name, value)
    local cvar_ptr = cvar.find(name)
    if cvar_ptr ~= 0 then
        if original_cvars[name] == nil then
            original_cvars[name] = { type = "int", val = cvar.get_int(cvar_ptr) }
        end
        cvar.set_int(cvar_ptr, value)
    end
end

local function SetF(name, value)

end

-- Cache for volatile CVars
local _cv_picmip = nil

local function EnforceVolatile()
    if _cv_picmip == nil then
        _cv_picmip = cvar.find("mat_picmip")
    end
    
    if _cv_picmip ~= 0 then
        local current_value = cvar.get_int(_cv_picmip)
        if current_value ~= 4 then
            SetI("mat_picmip", 4)
        end
    end
end

-- Restore original values on unload
function on_unload()
    local restored_count = 0

    for name, data in pairs(original_cvars) do
        local cvar_ptr = cvar.find(name)
        if cvar_ptr ~= 0 then
            if data.type == "int" then
                cvar.set_int(cvar_ptr, data.val)
            elseif data.type == "float" then
                cvar.set_float(cvar_ptr, data.val)
            elseif data.type == "string" then
                cvar.set_string(cvar_ptr, data.val)
            end
            restored_count = restored_count + 1
        end
    end

    client.print(string.format("FPS boost disabled. Successfully restored %d original CVars.", restored_count))
end
