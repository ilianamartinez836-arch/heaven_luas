local config = {
    enabled = true,
    aimkey = keys.MOUSE5,
    fov = 180.0,
    smooth = 1.0,
    max_distance = 4096.0,
    target_head = true,
    silent_aim = false,
    vischeck = true,
    auto_tank_hit = false,
    auto_secondary = false,
}

local locked_entity = nil

local function normalizeAngle(a)
    while a > 180 do a = a - 360 end
    while a < -180 do a = a + 360 end
    return a
end

local function calcAngle(src, dst)
    local dx = dst.x - src.x
    local dy = dst.y - src.y
    local dz = dst.z - src.z
    local hyp = math.sqrt(dx * dx + dy * dy)
    if hyp < 0.001 then hyp = 0.001 end
    return vector.new(math.deg(math.atan2(-dz, hyp)), math.deg(math.atan2(dy, dx)), 0)
end

local function getFOV(view, aim)
    local dx = normalizeAngle(aim.x - view.x)
    local dy = normalizeAngle(aim.y - view.y)
    return math.sqrt(dx * dx + dy * dy)
end

local function smoothAngles(cur, target, smooth)
    if not smooth or smooth <= 1.0 then return target end
    local f = 1.0 / smooth
    return vector.new(
        cur.x + normalizeAngle(target.x - cur.x) * f,
        cur.y + normalizeAngle(target.y - cur.y) * f,
        0
    )
end

local function is_visible(player)
    if not config.vischeck then return true end
    local ok, raw = pcall(entity.get_by_index_raw, player:get_index())
    if not ok or not raw or raw == 0 then return true end
    local ok2, dorm = pcall(entity.is_dormant, raw)
    if not ok2 then return true end
    return not dorm
end

local function get_aim_pos(player)
    if config.target_head then
        local ok, eye = pcall(function() return player:get_eye_position() end)
        if ok and eye and eye.x then return eye end
    end
    local ok2, o = pcall(function() return player:get_origin() end)
    if ok2 and o then
        return vector.new(o.x, o.y, o.z + 40.0)
    end
    return nil
end

local function is_valid_locked(ent, eye_pos)
    if not ent then return false end
    local ok_a, alive = pcall(function() return ent:is_alive() end)
    if not ok_a or not alive then return false end
    local ok_t, team = pcall(function() return ent:get_team() end)
    if not ok_t or team ~= 2 then return false end
    if ent.is_incapacitated then
        local ok_i, inc = pcall(function() return ent:is_incapacitated() end)
        if ok_i and inc then return false end
    end
    local ok_o, o = pcall(function() return ent:get_origin() end)
    if not ok_o or not o then return false end
    if vector.distance(eye_pos, o) > config.max_distance then return false end
    if not is_visible(ent) then return false end
    return true
end

local function getBestTarget(local_player)
    local eye_pos = local_player:get_eye_position()
    if not eye_pos then return nil end
    local view_angs = local_player:get_eye_angles()
    if not view_angs then view_angs = vector.new(0, 0, 0) end
    local local_idx = local_player:get_index()
    local mc = engine.get_max_clients() or 32
    local best = nil
    local best_fov = config.fov

    for i = 1, mc do
        local player = entity.get_by_index(i)
        if not player then goto continue end
        if i == local_idx then goto continue end
        local ok_a, alive = pcall(function() return player:is_alive() end)
        if not ok_a or not alive then goto continue end
        local ok_t, team = pcall(function() return player:get_team() end)
        if not ok_t or team ~= 2 then goto continue end
        if player.is_incapacitated then
            local ok_i, inc = pcall(function() return player:is_incapacitated() end)
            if ok_i and inc then goto continue end
        end
        if not is_visible(player) then goto continue end

        local aim_pos = get_aim_pos(player)
        if not aim_pos then goto continue end
        local dist = vector.distance(eye_pos, player:get_origin())
        if dist > config.max_distance then goto continue end

        local aim_angle = calcAngle(eye_pos, aim_pos)
        local fov = getFOV(view_angs, aim_angle)
        if fov > best_fov then goto continue end

        best_fov = fov
        best = { player = player, aim_pos = aim_pos, aim_angle = aim_angle }
        ::continue::
    end
    return best
end

function on_create_move(cmd, local_player)

            locked_entity = nil
            return
        end
    end

        locked_entity = nil

        return

        locked_entity = nil
    end

    if not locked_entity then
        local t = getBestTarget(local_player)
        if not t then return end
        locked_entity = t.player
    end

        final = aim_angle

    final.y = normalizeAngle(final.y)
    final.z = 0

    cmd:set_viewangles(final)
    if not config.silent_aim then
        engine.set_view_angles(final)

-- world_to_screen
local function w2s(x, y, z)
    local ok, mat_ptr = pcall(engine.world_to_screen_matrix)
    if not ok or not mat_ptr or mat_ptr == 0 then return nil end
    local function rm(i)
        local ok2, v = pcall(memory.read_float, mat_ptr + i * 4)
        return ok2 and v or 0
    end
    local m00, m01, m02, m03 = rm(0), rm(1), rm(2), rm(3)
    local m10, m11, m12, m13 = rm(4), rm(5), rm(6), rm(7)
    local m30, m31, m32, m33 = rm(12), rm(13), rm(14), rm(15)
    local w = m30 * x + m31 * y + m32 * z + m33
    if w < 0.001 then return nil end
    local inv = 1.0 / w
    local sw, sh = 0, 0
    pcall(function() sw, sh = engine.get_screen_size() end)
    if not sw or sw == 0 then return nil end
    local sx = (m00 * x + m01 * y + m02 * z + m03) * inv
    local sy = (m10 * x + m11 * y + m12 * z + m13) * inv
    return math.floor(sw / 2 + sx * sw / 2),
           math.floor(sh / 2 - sy * sh / 2)
end

-- scan SOLO de jugadores
local function scan_specs()
    local now = os.clock()
    if now - last_scan < SCAN_RATE then return end
    last_scan = now

    spec_cache = {}
    local max_clients = 32
    pcall(function() max_clients = engine.get_max_clients() end)
    if not max_clients or max_clients <= 0 or max_clients > 64 then max_clients = 32 end

    -- iterar SOLO los slots de jugador (1..max_clients)
    for i = 1, max_clients do
        local e
        pcall(function() e = entity.get_by_index(i) end)
        if e then
            -- validar que sea un jugador real: existe player_info
            local info
            pcall(function() info = engine.get_player_info(i) end)
            if info and info.name and info.name ~= "" then
                local mode
                pcall(function() mode = e:get_observer_mode() end)
                if mode and mode > 0 then
                    table.insert(spec_cache, { ent = e, idx = i, name = info.name })
                end
            end
        end
    end
end

-- language: Lua, file: rage_strafer.lua, target: Heaven Hook (L4D2)
-- Versión agresiva pero estable. Sin extra_commands, sin speed_bonus, sin overcorrect.
-- Primero confirmamos que esto no crashea, después subimos.

local IN_JUMP          = 2
local MOVETYPE_WALK    = 2
local MOVETYPE_NOCLIP  = 8
local MOVETYPE_LADDER  = 9
local MOVETYPE_OBSERVER = 10

local MAX_SPEED  = 250.0
local AIR_ACCEL  = 300.0
local WISH_SPEED = 300.0
local MAXMOVE    = 450.0

local function norm_rad(a)
    a = a % (2 * math.pi)
    if a >  math.pi then a = a - 2 * math.pi end
    if a < -math.pi then a = a + 2 * math.pi end
    return a
end
local function angle_diff_rad(a1, a2) return norm_rad(a1 - a2) end

local function apply_bhop(cmd, lp)
    local b = cmd:get_buttons()
    if (b & IN_JUMP) ~= 0 and not lp:is_on_ground() then
        cmd:set_buttons(b & ~IN_JUMP)
    end
end

local function apply_strafe(cmd, lp)
    local mt = lp:get_move_type()
    if mt == MOVETYPE_NOCLIP or mt == MOVETYPE_LADDER or mt == MOVETYPE_OBSERVER then return end
    if lp:is_on_ground() then return end

    local vel = lp:get_velocity()
    if not vel then return end
    local speed = math.sqrt(vel.x * vel.x + vel.y * vel.y)
    if speed < 2.0 then return end

    local term = (WISH_SPEED / AIR_ACCEL) / MAX_SPEED * 100.0 / speed
    local perfect_delta = 0.0
    if term > -1.0 and term < 1.0 then
        perfect_delta = math.acos(term)
    else
        return   -- fuera de rango: no aplicamos strafe este tick
    end

    local va = cmd:get_viewangles()
    if not va then return end

    local yaw_rad = math.rad(va.y)
    local vel_dir  = math.atan2(vel.y, vel.x) - yaw_rad
    local wish_ang = math.atan2(-cmd:get_sidemove(), cmd:get_forwardmove())
    local delta    = angle_diff_rad(vel_dir, wish_ang)
    local move_dir = delta < 0.0 and (vel_dir + perfect_delta) or (vel_dir - perfect_delta)

    -- si move_dir no es finito, skip
    if move_dir ~= move_dir or move_dir == math.huge or move_dir == -math.huge then return end

    cmd:set_forwardmove(math.cos(move_dir) * MAXMOVE)
    cmd:set_sidemove(-math.sin(move_dir) * MAXMOVE)
end

function on_create_move(cmd, local_player)
    if not cmd or not local_player then return end
    if not local_player:is_alive() then return end
    if local_player:get_move_type() ~= MOVETYPE_WALK then return end

    local b = cmd:get_buttons()
    if (b & IN_JUMP) ~= 0 then
        apply_bhop(cmd, local_player)
        apply_strafe(cmd, local_player)
    end
end

local KEY = keys.KEY_K
local IN_ATTACK = 1

local cfg = {
    enabled      = false,
    spin_speed   = 15.0,
    pitch        = -89.0,
    fix_movement = true,   -- nueva opcion, ON por default
}

local spin_yaw  = 0.0
local prev_down = false

local function log(s) pcall(client.print, "[spinbot] " .. tostring(s)) end

local function norm_yaw(y)
    y = y % 360
    if y > 180 then y = y - 360 end
    if y < -180 then y = y + 360 end
    return y
end

-- rota el forward/side por -delta para compensar la rotacion del yaw
local function fix_movement(cmd, old_yaw, new_yaw)
    if not cfg.fix_movement then return end

    local delta_rad = math.rad(new_yaw - old_yaw)
    local c = math.cos(delta_rad)
    local s = math.sin(delta_rad)

    local fm, sm = 0, 0
    pcall(function()
        fm = cmd:get_forwardmove()
        sm = cmd:get_sidemove()
    end)

    if fm == 0 and sm == 0 then return end

    -- aplicar rotacion inversa
    local new_fm = fm * c - sm * s
    local new_sm = fm * s + sm * c

    pcall(function()
        cmd:set_forwardmove(new_fm)
        cmd:set_sidemove(new_sm)
    end)
end

function on_create_move(cmd, local_player)
    if not cmd then return end

    local down = false
    pcall(function() down = input.is_key_down(KEY) end)
    if down and not prev_down then
        cfg.enabled = not cfg.enabled
        log(cfg.enabled and "ON" or "OFF")
    end
    prev_down = down

    if not cfg.enabled then return end

    local buttons = 0
    pcall(function() buttons = cmd:get_buttons() end)
    if (buttons & IN_ATTACK) ~= 0 then return end

    -- guardar el yaw viejo antes de modificarlo
    local old_yaw = 0.0
    pcall(function()
        local va = cmd:get_viewangles()
        if va and va.y then old_yaw = va.y end
    end)

    -- avanzar spin
    spin_yaw = norm_yaw(spin_yaw + cfg.spin_speed)

    -- aplicar angulos nuevos
    pcall(function()
        cmd:set_viewangles(vector.new(cfg.pitch, spin_yaw, 0))
    end)

    -- corregir movimiento
    fix_movement(cmd, old_yaw, spin_yaw)
end

local nofall = {
    original_state = true,
    last_team = 0
}

function on_create_move(cmd, local_player)
    if not local_player then return end
    if not engine.is_in_game() then return end

    local current_team = local_player:get_team()
    
    if current_team == nofall.last_team then
        return
    end

    nofall.last_team = current_team

    if current_team == 3 then
        -- Infectado → Desactivar No Fall Damage
        nofall.original_state = cvar_vars.get_bool("vars_enable_no_fall")
        cvar_vars.set_bool("vars_enable_no_fall", false)
    elseif current_team == 2 then
        -- Survivor → Restaurar estado original
        cvar_vars.set_bool("vars_enable_no_fall", nofall.original_state)
    end
end

function on_unload()
    cvar_vars.set_bool("vars_enable_no_fall", nofall.original_state)
end

local ffi = rawget(_G, "ffi")
if not ffi then return end

local function p(msg)
    pcall(function() client.print(tostring(msg) .. "\n") end)
end

local function band(a, b)
    local res, bitval = 0, 1
    a, b = a or 0, b or 0
    for _ = 0, 31 do
        if (a % 2 == 1) and (b % 2 == 1) then res = res + bitval end
        a = math.floor(a / 2)
        b = math.floor(b / 2)
        bitval = bitval * 2
        if a == 0 and b == 0 then break end
    end
    return res
end

local function r32(addr)
    local ok, v = pcall(memory.read_int, addr)
    if ok then return v end
    return nil
end

local function read_cstr(addr, n)
    n = n or 32
    if not addr or addr == 0 then return nil end
    local t = {}
    for i = 0, n - 1 do
        local w = r32(addr + i)
        if not w then break end
        local b = band(w, 0xFF)

local function calc_angle(src, dst)
    local dx = dst.x - src.x
    local dy = dst.y - src.y
    local dz = dst.z - src.z
    local hyp = math.sqrt(dx * dx + dy * dy)
    local pitch = math.deg(math.atan2(-dz, hyp))
    local yaw   = math.deg(math.atan2(dy, dx))
    return vector.new(pitch, yaw, 0)

local function vec_length(v)
    return math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z)
end

-- Avance 1 tick de velocidad, clamp 16u (igual Byx)
local function eye_advance(lp, eye_real)
    local vel = lp:get_velocity()
    if not vel then return eye_real end

    local ax = vel.x * TICK_INTERVAL
    local ay = vel.y * TICK_INTERVAL
    local az = vel.z * TICK_INTERVAL
    local len = math.sqrt(ax * ax + ay * ay + az * az)

    if len > MAX_ADVANCE and len > 0 then
        local s = MAX_ADVANCE / len
        ax, ay, az = ax * s, ay * s, az * s
    end

    return vector.new(eye_real.x + ax, eye_real.y + ay, eye_real.z + az)
end

                              + (dz * inv) * forward.z
                    if dot >= cos_cone then
                        best_ent  = ent
                        best_pos  = pos
                        best_dist = dist
                        best_idx  = i
                    end
                end
            end
        end
    end

    if not best_ent then return end
    if not is_visible(eye, best_pos, best_idx) then return end

    if cfg.aim_at_target then
        local ang = calc_angle(eye, best_pos)
        ang.x = clamp_pitch(ang.x)
        ang.y = normalize_angle(ang.y)

        -- Silent move fix (Byx): re-proyectar forward/side respecto al nuevo yaw
        local fm = cmd:get_forwardmove()
        local sm = cmd:get_sidemove()
        if fm ~= 0 or sm ~= 0 then
            local yd = math.rad(ang.y - view.y)
            local c, s = math.cos(yd), math.sin(yd)
            cmd:set_forwardmove(fm * c - sm * s)
            cmd:set_sidemove(fm * s + sm * c)
        end

        cmd:set_viewangles(ang)
        -- No engine.set_view_angles → silent shove (cámara no salta)
    end

    local buttons = cmd:get_buttons()
    cmd:set_buttons(buttons | IN_ATTACK2)

    stats.shoves = stats.shoves + 1
    stats.last_dist = best_dist
end

local PI = 3.14159

-- Helpers del creador
local function normalize_angle(angle)
    while angle > 180 do angle = angle - 360 end
    while angle < -180 do angle = angle + 360 end
    return angle
end

local function calc_angle(src, dst)
    local delta = vector.new(dst.x - src.x, dst.y - src.y, dst.z - src.z)
    local hyp = math.sqrt(delta.x^2 + delta.y^2)
    local pitch = math.deg(math.atan2(-delta.z, hyp))
    local yaw = math.deg(math.atan2(delta.y, delta.x))
    return vector.new(pitch, yaw, 0)
end

local function get_fov(view_angles, aim_angles)
    local dx = normalize_angle(aim_angles.x - view_angles.x)
    local dy = normalize_angle(aim_angles.y - view_angles.y)
    return math.sqrt(dx*dx + dy*dy)
end

        local ent = entity.get_by_index(i)
        if ent and ent:get_index() ~= local_player:get_index() then
            local zclass = ent:get_zombie_class()
            if ent:is_alive() and ent:get_team() == 3 and not ent:is_ghost() and
               ((zclass == 3 and client.get_shared_bool("as_hunter", true)) or
                (zclass == 5 and client.get_shared_bool("as_jockey", true)) or
                (zclass == 1 and client.get_shared_bool("as_smoker", true)) or
                (zclass == 4 and client.get_shared_bool("as_spitter", true)) or
                (zclass == 2 and client.get_shared_bool("as_boomer", true))) then
                local target_pos = vector.new(ent:get_origin().x, ent:get_origin().y, ent:get_origin().z + 55)
                local dist = vector.distance(eye_pos, target_pos)
                if dist <= range then
                    local aim_angles = calc_angle(eye_pos, target_pos)
                    local this_fov = legit and get_fov(view_angles, aim_angles) or 0
                    if this_fov < best_fov then
                        best_fov = this_fov
                        best_target = ent
                    end
                end
            end
        end
    end
    
    if best_target then
        cmd:set_buttons(buttons | 2048) -- IN_ATTACK2
        last_shove_time = curtime  -- reset delay real
        
        local target_pos = vector.new(best_target:get_origin().x, best_target:get_origin().y, best_target:get_origin().z + 55)

            -- NO engine.set_view_angles → tu cámara no gira
        else
            cmd:set_viewangles(aim_angles)
            engine.set_view_angles(aim_angles) -- Snap completo
        end
    end
end

-- Jockey Free Ride (Heaven)
-- Cuando TÚ eres Jockey y estás cabalgando: control total de movimiento
-- (la resistencia del survivor se neutraliza lo máximo posible en cliente)

local JOCKEY_CLASS = 5
local TEAM_INFECTED = 3
local IN_FORWARD = 8
local IN_BACK = 16
local IN_MOVELEFT = 512
local IN_MOVERIGHT = 1024
local IN_SPEED = 131072

-- offsets candidatos de victim handle (validar en tu build si falla detección)
local VICTIM_HANDLE_OFFSETS = { 10060, 10056, 10064, 10052, 16020, 16024 }

local function is_jockey(lp)
    if not lp or not lp:is_alive() then return false end
    if lp:get_team() ~= TEAM_INFECTED then return false end
    if lp:is_ghost and lp:is_ghost() then return false end
    return lp:get_zombie_class() == JOCKEY_CLASS
end

local function handle_valid(h)
    if not h or h == 0 or h == -1 then return false end
    -- handles Source suelen ser serializados; índice bajo en low bits
    local idx = h % 65536
    return idx > 0 and idx < 2048
end

local function is_riding(lp)
    -- 1) sequence name
    local ok, seq = pcall(function()
        local s = lp:get_sequence_name and lp:get_sequence_name(lp:get_solid_flags and 0 or 0)
        return s
    end)
    -- mejor: intentar get_sequence si existe vía raw
    local raw = entity.get_by_index_raw(lp:get_index())
    if raw and raw ~= 0 then
        local ok2, seq_id = pcall(entity.get_sequence_from_ptr, raw)
        if ok2 and seq_id then
            local ok3, name = pcall(function() return lp:get_sequence_name(seq_id) end)
            if ok3 and name then
                local n = string.lower(name)
                if string.find(n, "jockey", 1, true)
                    or string.find(n, "ride", 1, true)
                    or string.find(n, "leap", 1, true) then
                    -- leap solo no basta; ride/jockey sí
                    if string.find(n, "ride", 1, true) or string.find(n, "jockey", 1, true) then
                        return true
                    end
                end
            end
        end

        -- 2) victim handle en memoria
        for _, off in ipairs(VICTIM_HANDLE_OFFSETS) do
            local okh, h = pcall(memory.read_int, raw + off)
            if okh and handle_valid(h) then
                local idx = h % 65536
                local vic = entity.get_by_index(idx)
                if vic and vic:is_alive() and vic:get_team() == 2 then
                    return true
                end
            end
        end
    end

    -- 3) fallback: move_type / flags raros mientras jockey (menos fiable)
    return false
end

local function force_full_move(cmd)
    local buttons = cmd:get_buttons()
    local fm = 0.0
    local sm = 0.0

    -- lee input real
    if (buttons & IN_FORWARD) ~= 0 then fm = fm + 1 end
    if (buttons & IN_BACK) ~= 0 then fm = fm - 1 end
    if (buttons & IN_MOVERIGHT) ~= 0 then sm = sm + 1 end
    if (buttons & IN_MOVELEFT) ~= 0 then sm = sm - 1 end

    -- también respeta analog stick de cmd si el jugador ya movió
    local cf = cmd:get_forwardmove()
    local cs = cmd:get_sidemove()
    if math.abs(cf) > 1 then fm = (cf > 0) and 1 or -1 end
    if math.abs(cs) > 1 then sm = (cs > 0) and 1 or -1 end

    -- si no hay input, empuja siempre adelante (control total tipo zonemod)
    if fm == 0 and sm == 0 then
        fm = 1
    end

    -- normaliza a máximo wish speed
    local len = math.sqrt(fm * fm + sm * sm)
    if len > 0 then
        fm = (fm / len) * 450.0
        sm = (sm / len) * 450.0
    end

    cmd:set_forwardmove(fm)
    cmd:set_sidemove(sm)

    -- limpia back implícito: solo deja flags coherentes
    local nb = buttons
    if fm > 0 then
        nb = (nb | IN_FORWARD) & ~IN_BACK
    elseif fm < 0 then
        nb = (nb | IN_BACK) & ~IN_FORWARD
    end
    if sm > 0 then
        nb = (nb | IN_MOVERIGHT) & ~IN_MOVELEFT
    elseif sm < 0 then
        nb = (nb | IN_MOVELEFT) & ~IN_MOVERIGHT
    end
    cmd:set_buttons(nb)
end

function on_create_move(cmd, local_player)
    if not cmd or not local_player then return end
    if not engine.is_in_game() then return end
    if not is_jockey(local_player) then return end
    if not is_riding(local_player) then return end

    force_full_move(cmd)
end

-- refuerzo en prediction (misma lógica)
function on_create_move_prediction(cmd, local_player)
    if not cmd or not local_player then return end
    if not is_jockey(local_player) then return end
    if not is_riding(local_player) then return end
    force_full_move(cmd)
end
